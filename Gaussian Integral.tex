\begin{center}
{\Large \textbf{
Simple Proof for the Area Under the Curve of the Normal Distribution
}}
\end{center}

\begin{center}
    Claim: $\displaystyle \int_{-\infty}^{\infty}\frac{e^{\frac{-x^2}{2}}}{\sqrt{2\pi}}dx=1$
\end{center}

\textit{Proof.} Assign the expression to a variable. 
\begin{center}
    Let $\displaystyle I =  \int_{-\infty}^{\infty}\frac{e^{\frac{-x^2}{2}}}{\sqrt{2\pi}}dx$
\end{center}

\begin{center}
    $\displaystyle \therefore I^2=\int_{-\infty}^{\infty}\frac{e^{\frac{-x^2}{2}}}{\sqrt{2\pi}}dx\int_{-\infty}^{\infty}\frac{e^{\frac{-y^2}{2}}}{\sqrt{2\pi}}dy$
\end{center}

Rewrite the expression

\begin{center}
    $\displaystyle I^2= \int_{-\infty}^{\infty}\int_{-\infty}^\infty \frac{e^{\frac{-x^2}{2}}}{\sqrt{2\pi}}\cdot\frac{e^{\frac{-y^2}{2}}}{\sqrt{2\pi}}dxdy$
\end{center}

Simplify

\begin{center}
    $\displaystyle I^2=\frac{1}{2\pi}\int_{-\infty}^{\infty}\int_{-\infty}^{\infty}e^\frac{-{(x^2+y^2)}}{2}dxdy$
\end{center}

Transform to polar coordinates. Refer to figure below: 

\begin{center}
        $\displaystyle I^2=\frac{1}{2\pi}\int_{0}^{2\pi}\int_0^{\infty}e^{-\frac{r^2}{2}} r \,drd\theta$
\end{center}

Perform u-substitution

\[
\text{Let } u=-\frac{r^2}{2}, \qquad \therefore\ du=-r\,dr \rightarrow dr=-\frac{du}{r}
\]

Substituting the values of $u$ and $du$ to the expression yields

\begin{center}
    $\displaystyle  I^2=-\frac{1}{2\pi}\int_{0}^{2\pi}\int_0^{\infty}e^ur\,\frac{du}{r}\,d\theta$
\end{center}

\newpage
Rewrite the intgeral

\begin{center}
    $\displaystyle I^2=-\frac{1}{2\pi}\int_0^{2\pi}\left(\lim_{n\rightarrow\infty}\int_0^ne^udu\right)d\theta$
\end{center}

\begin{center}
    $\displaystyle I^2=-\frac{1}{2\pi}\int_0^{2\pi}\left(\lim_{n\rightarrow\infty}e^{-\frac{r^2}{2}}\Big|_0^n\right)d\theta$
\end{center}

Evaluate the limit

\begin{center}
    $\displaystyle I^2=-\frac{1}{2\pi}\int_0^{2\pi}-1\,d\theta=\frac{1}{2\pi}\int_0^{2\pi}d\theta$
\end{center}

Solving the integral yields

\begin{center}
    $\displaystyle I^2=\frac{1}{2\pi}\cdot 2\pi=1$
\end{center}

Since $I>0$, taking the square root gives

\begin{center}
    $\displaystyle I=1$
\end{center}

\begin{center}
     $\displaystyle \therefore \int_{-\infty}^{\infty}\frac{e^{\frac{-x^2}{2}}}{\sqrt{2\pi}}dx=1$
\end{center}
